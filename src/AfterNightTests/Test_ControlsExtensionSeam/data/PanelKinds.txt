# Every kind a description of an extension's panel may hold, each field of it and the bounds a
# description is held to: what panelKinds() of controls/ExtensionPanelDescription.h declares.
# docs/dev/EXTENSION_PANEL_KINDS.md, the extension author's reference, is generated from this
# record by build/scripts/tools/panel_kinds_reference.py.
#
# limits      name | value | what it bounds
# parameters  the kinds of a process window's list; widgets, those of a tree of widgets:
#               kind | role | holds children | aliases | what it is
#                   field | what it holds | when absent | bound | what it does

[limits]
nodes | 512 | How many nodes a description may hold, groups and what they hold counted.
depth | 32 | How deep groups may nest.
short text | 256 | The characters of an id, a label, a title or a placeholder.
long text | 8192 | The characters of a prose text: a banner's message, a tooltip, a description.
console text | 262144 | The characters of a console's text.
entries of a choice | 1024 | The entries of a choice, a combo box or a button group.
entries of a list | 4096 | The entries of any other list: points, ranges, rows.
