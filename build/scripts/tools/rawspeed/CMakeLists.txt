cmake_minimum_required(VERSION 3.25)
project(AfterNightRawQualification LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(RawRuntimeProbe RawRuntimeProbe.cpp)
if(UNIX)
    target_link_libraries(RawRuntimeProbe PRIVATE dl)
endif()
install(TARGETS RawRuntimeProbe RUNTIME DESTINATION .)
get_filename_component(AN_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
if(WIN32)
    set(AN_GENERAL "${AN_REPO}/vcpkg/installed/x64-windows")
    set(AN_PRIVATE "${AN_REPO}/vcpkg/installed/x64-windows-raw")
else()
    set(AN_GENERAL "${AN_REPO}/vcpkg/installed/x64-linux")
    set(AN_PRIVATE "${AN_REPO}/vcpkg/installed/x64-linux-raw")
endif()
set(AN_PRIVATE_SOURCE "${AN_PRIVATE}/share/afternight-libraw/source")
if(AN_PRIVATE_PREFIX)
    # A local sanitizer build supplies instrumented private archives without replacing vcpkg.
    set(AN_PRIVATE "${AN_PRIVATE_PREFIX}")
    # vcpkg prepends its own prefix; CMAKE_PREFIX_PATH alone does not select these packages.
    foreach(package AfterNightLibRaw AfterNightRawSpeed)
        if(package STREQUAL "AfterNightLibRaw")
            set(port afternight-libraw)
        else()
            set(port afternight-rawspeed)
        endif()
        set(${package}_DIR "${AN_PRIVATE}/share/${port}")
        if(NOT EXISTS "${${package}_DIR}/${package}Config.cmake")
            message(FATAL_ERROR "Missing private qualification package: ${${package}_DIR}")
        endif()
    endforeach()
endif()
option(AN_ENGINE_DRIVERS "Build the end-to-end engine benchmark drivers" ON)
option(AN_NATIVE_ONLY "Build rollback qualification without either private package" OFF)
if(AN_ENGINE_DRIVERS)
    add_executable(RawEngineBench RawEngineBench.cpp)
    target_include_directories(RawEngineBench PRIVATE "${AN_REPO}/src/AfterNightEngine/include")
    foreach(target RawEngineBench)
        target_include_directories(${target} PRIVATE "${AN_GENERAL}/include")
        if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
            target_compile_options(${target} PRIVATE -Werror=dangling-gsl)
        endif()
        if(MSVC)
            target_compile_options(${target} PRIVATE /utf-8 /bigobj /Zc:__cplusplus)
            target_compile_definitions(${target} PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
            target_link_libraries(${target} PRIVATE psapi)
        endif()
    endforeach()
    if(WIN32)
        target_link_libraries(RawEngineBench PRIVATE "${AN_REPO}/lib/msvc/AfterNightEngine$<$<CONFIG:Debug>:_d>.lib")
    else()
        target_link_libraries(RawEngineBench PRIVATE "${AN_REPO}/bin/clang/$<LOWER_CASE:$<CONFIG>>/libAfterNightEngine$<$<CONFIG:Debug>:_d>.so")
    endif()
    install(TARGETS RawEngineBench RUNTIME DESTINATION .)
endif()
if(NOT AN_NATIVE_ONLY)
list(PREPEND CMAKE_PREFIX_PATH "${AN_PRIVATE}")
find_package(AfterNightLibRaw CONFIG REQUIRED)
if(WIN32)
    add_library(RawSpeedFault SHARED FaultBackend.cpp)
    target_include_directories(RawSpeedFault PRIVATE "${AN_PRIVATE}/include/afternight-rawspeed")
    target_compile_definitions(RawSpeedFault PRIVATE AN_RAWSPEED_BUILD _CRT_SECURE_NO_WARNINGS)
    set_target_properties(RawSpeedFault PROPERTIES OUTPUT_NAME afternight_rawspeed DEBUG_POSTFIX d
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fault")
endif()
add_executable(RawPrivateOracle RawOracle.cpp)
target_compile_definitions(RawPrivateOracle PRIVATE AN_PRIVATE_ORACLE)
target_include_directories(RawPrivateOracle BEFORE PRIVATE "${AN_PRIVATE}/include")
target_link_libraries(RawPrivateOracle PRIVATE AfterNight::afternight_libraw_r)
install(TARGETS RawPrivateOracle RUNTIME DESTINATION .)
if(UNIX)
    # The installed source exposes the protected reader to a fixture-free UBSan regression.
    add_executable(RawNativeTests RawNativeTests.cpp)
    target_include_directories(RawNativeTests BEFORE PRIVATE "${AN_PRIVATE_SOURCE}")
    target_compile_definitions(RawNativeTests PRIVATE LIBRAW_LIBRARY_BUILD LIBRAW_NODLL)
    target_link_libraries(RawNativeTests PRIVATE AfterNight::afternight_libraw_r)
    install(TARGETS RawNativeTests RUNTIME DESTINATION .)
endif()
endif()
add_executable(RawStockOracle RawOracle.cpp)
foreach(target RawStockOracle RawPrivateOracle)
  if(TARGET ${target})
    target_include_directories(${target} PRIVATE "${AN_GENERAL}/include")
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8)
        target_link_libraries(${target} PRIVATE dbghelp)
    endif()
  endif()
endforeach()
install(TARGETS RawStockOracle RUNTIME DESTINATION .)
if(WIN32)
    target_link_libraries(RawStockOracle PRIVATE "$<$<CONFIG:Debug>:${AN_GENERAL}/debug/lib/raw_rd.lib>" "$<$<NOT:$<CONFIG:Debug>>:${AN_GENERAL}/lib/raw_r.lib>")
else()
    target_compile_definitions(RawStockOracle PRIVATE LIBRAW_NODLL)
    target_link_directories(RawStockOracle PRIVATE "${AN_GENERAL}/$<$<CONFIG:Debug>:debug/>lib")
    target_link_libraries(RawStockOracle PRIVATE raw_r jpeg lcms2 z m)
endif()
