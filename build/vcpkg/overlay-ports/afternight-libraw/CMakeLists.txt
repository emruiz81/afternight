cmake_minimum_required(VERSION 3.25)
project(AfterNightLibRaw VERSION 0.22.1 LANGUAGES C CXX)
include(GNUInstallDirs)
find_package(AfterNightRawSpeed CONFIG REQUIRED)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ENABLE_OPENMP OFF CACHE BOOL "" FORCE)
set(ENABLE_RAWSPEED OFF CACHE BOOL "" FORCE)
set(ENABLE_X3FTOOLS ON CACHE BOOL "" FORCE)
set(ENABLE_LCMS ON CACHE BOOL "" FORCE)
set(LIBRAW_INSTALL OFF CACHE BOOL "" FORCE)
set(CMAKE_REQUIRE_FIND_PACKAGE_JPEG ON)
set(CMAKE_REQUIRE_FIND_PACKAGE_ZLIB ON)
add_subdirectory("${RAW_SOURCE}" libraw EXCLUDE_FROM_ALL)
add_custom_target(afternight_libraw ALL DEPENDS raw_r)
set_target_properties(raw_r PROPERTIES OUTPUT_NAME afternight_libraw_r COMPILE_PDB_NAME afternight_libraw_r
    EXPORT_NAME afternight_libraw_r DEBUG_POSTFIX d
    INTERFACE_INCLUDE_DIRECTORIES "$<INSTALL_INTERFACE:include/afternight-libraw>")
target_compile_definitions(raw_r PRIVATE USE_RAWSPEED3 USE_RAWSPEED_BITS)
target_link_libraries(raw_r PRIVATE AfterNight::afternight_rawspeed)
if(UNIX)
    target_link_options(raw_r PRIVATE "LINKER:--exclude-libs,ALL")
endif()
install(TARGETS raw_r EXPORT AfterNightLibRawTargets)
install(DIRECTORY "${RAW_SOURCE}/libraw/" DESTINATION include/afternight-libraw/libraw FILES_MATCHING PATTERN "*.h")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/libraw/libraw_config.h" DESTINATION include/afternight-libraw/libraw)
install(EXPORT AfterNightLibRawTargets NAMESPACE AfterNight:: DESTINATION share/afternight-libraw)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/AfterNightLibRawConfig.cmake"
    "include(CMakeFindDependencyMacro)\nfind_dependency(AfterNightRawSpeed CONFIG)\nfind_dependency(lcms2 CONFIG)\nfind_dependency(JPEG)\nfind_dependency(ZLIB)\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/AfterNightLibRawTargets.cmake\")\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/AfterNightLibRawConfig.cmake" DESTINATION share/afternight-libraw)
