cmake_minimum_required(VERSION 3.25)
project(AfterNightRawSpeed VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
find_package(pugixml CONFIG REQUIRED)
find_package(JPEG REQUIRED)
find_package(ZLIB REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(RAWSPEED_CACHELINESIZE 64)
set(RAWSPEED_PAGESIZE 4096)
set(WITH_SSE2 ON)
set(HAVE_PUGIXML ON)
set(HAVE_JPEG ON)
set(HAVE_JPEG_MEM_SRC ON)
set(HAVE_ZLIB ON)
set(HAVE_CXX_THREAD_LOCAL ON)
set(HAVE_OPENMP OFF)
if(WIN32)
    set(HAVE_ALIGNED_MALLOC ON)
else()
    set(HAVE_POSIX_MEMALIGN ON)
endif()
configure_file("${RAW_SOURCE}/src/config.h.in" "${CMAKE_CURRENT_BINARY_DIR}/rawspeedconfig.h")
file(READ "${RAW_SOURCE}/data/cameras.xml" CAMERA_XML)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/Cameras.cpp" "const char* anRawSpeedCameraXml = R\"AN_CAMERA(${CAMERA_XML})AN_CAMERA\";\n")
file(GLOB_RECURSE RAW_SOURCES CONFIGURE_DEPENDS "${RAW_SOURCE}/src/librawspeed/*.cpp")
add_library(rawspeed_objects OBJECT ${RAW_SOURCES} "${CMAKE_CURRENT_BINARY_DIR}/Cameras.cpp")
add_library(afternight_rawspeed AnRawSpeed.cpp $<TARGET_OBJECTS:rawspeed_objects>)
foreach(target rawspeed_objects afternight_rawspeed)
target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
    "${RAW_SOURCE}/src/librawspeed" "${RAW_SOURCE}/src/external")
target_include_directories(afternight_rawspeed PUBLIC "$<INSTALL_INTERFACE:include/afternight-rawspeed>")
target_compile_definitions(${target} PRIVATE AN_RAWSPEED_BUILD _CRT_SECURE_NO_WARNINGS)
if(WIN32)
    # MSVC's debug std::string allocates a proxy in its noexcept default constructor.
    # Keep the debug CRT and assertions, but avoid an uncatchable OOM in that constructor.
    target_compile_definitions(${target} PRIVATE _ITERATOR_DEBUG_LEVEL=0)
endif()
target_link_libraries(${target} PRIVATE pugixml::pugixml JPEG::JPEG ZLIB::ZLIB)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(${target} PRIVATE -ffp-contract=off)
endif()
endforeach()
if(NOT BUILD_SHARED_LIBS)
    target_compile_definitions(afternight_rawspeed PUBLIC AN_RAWSPEED_STATIC)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(afternight_rawspeed PRIVATE -ffp-contract=off)
endif()
if(UNIX)
    target_link_options(afternight_rawspeed PRIVATE "LINKER:--exclude-libs,ALL")
endif()
set_target_properties(afternight_rawspeed PROPERTIES DEBUG_POSTFIX d)
option(AN_RAWSPEED_TESTS "Build the boundary, allocation and exact-curve tests" ON)
if(AN_RAWSPEED_TESTS)
    add_executable(AnRawSpeedTests AnRawSpeedTests.cpp $<TARGET_OBJECTS:rawspeed_objects>)
    target_include_directories(AnRawSpeedTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}"
        "${RAW_SOURCE}/src/librawspeed" "${RAW_SOURCE}/src/external")
    target_compile_definitions(AnRawSpeedTests PRIVATE AN_RAWSPEED_STATIC _CRT_SECURE_NO_WARNINGS NOMINMAX)
    if(WIN32)
        target_compile_definitions(AnRawSpeedTests PRIVATE _ITERATOR_DEBUG_LEVEL=0)
    endif()
    target_link_libraries(AnRawSpeedTests PRIVATE pugixml::pugixml JPEG::JPEG ZLIB::ZLIB)
    enable_testing()
    add_test(NAME RawSpeedBoundary COMMAND AnRawSpeedTests)
    install(TARGETS AnRawSpeedTests RUNTIME DESTINATION tools/afternight-rawspeed)
endif()
install(TARGETS afternight_rawspeed EXPORT AfterNightRawSpeedTargets)
install(FILES AnRawSpeed.h DESTINATION include/afternight-rawspeed)
install(EXPORT AfterNightRawSpeedTargets NAMESPACE AfterNight:: DESTINATION share/afternight-rawspeed)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/AfterNightRawSpeedConfig.cmake"
    "include(CMakeFindDependencyMacro)\nfind_dependency(pugixml CONFIG)\nfind_dependency(JPEG)\nfind_dependency(ZLIB)\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/AfterNightRawSpeedTargets.cmake\")\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/AfterNightRawSpeedConfig.cmake" DESTINATION share/afternight-rawspeed)
